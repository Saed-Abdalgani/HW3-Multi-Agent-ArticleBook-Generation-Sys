% --- Decorated equation (FR-9, FR-10) ---
\begin{equation}
\label{eq:normalized}
\mathbf{z}^{(t+1)} =
\begin{bmatrix}
\cos\theta & -\sin\theta \\
\sin\theta & \cos\theta
\end{bmatrix}
\mathbf{z}^{(t)}
+
\begin{cases}
\eta \,\nabla \mathcal{L}(\mathbf{z}^{(t)}), &
\lVert \nabla \mathcal{L}(\mathbf{z}^{(t)}) \rVert \le \tau, \\[6pt]
\eta \,\dfrac{\nabla \mathcal{L}(\mathbf{z}^{(t)})}
{\lVert \nabla \mathcal{L}(\mathbf{z}^{(t)}) \rVert}, &
\text{otherwise.}
\end{cases}
\end{equation}
